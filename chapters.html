เริ่มบทเรียน →
InputOutput.java 
import java.util.Scanner;

class InputOutput {

    public static void main(
        String[] args
    ) {

        Scanner sc =
            new Scanner(System.in);

        System.out.printf(
            "ชื่อ: "
        );

        String name =
            sc.nextLine();

        System.out.println(name);

    }

}
